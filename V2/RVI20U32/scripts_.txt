


.section .text
.globl _start
.align 2

_start:
    # ============================================================
    # INIT
    # ============================================================
    addi x1,  x0, 64       # base A
    addi x2,  x0, 128      # base B
    addi x3,  x0, 192      # base C
    addi x4,  x0, 7
    addi x5,  x0, 32
    addi x6,  x0, 96

    # ============================================================
    # 1) STORE immediate producer -> store_data forwarding
    # ============================================================
    addi x10, x0, 25
    sw   x10, 0(x1)        # MEM[64] = 25
    lw   x11, 0(x1)        # x11 = 25
    add  x12, x11, x11     # x12 = 50, load-use

    # ============================================================
    # 2) Load-use + store-data load forwarding
    # ============================================================
    lw   x13, 0(x1)        # x13 = 25
    sw   x13, 4(x1)        # MEM[68] = 25
    lw   x14, 4(x1)        # x14 = 25
    addi x15, x14, 7       # x15 = 32

    # ============================================================
    # 3) Branch RAW: add produces x16, branch consumes x16
    # ============================================================
    add  x16, x15, x0      # x16 = 32
    beq  x16, x5, B1_OK    # must be taken using forwarding
    addi x20, x0, 111      # WRONG PATH
    addi x21, x0, 222      # WRONG PATH

B1_OK:
    # ============================================================
    # 4) Store branch result, load back, load-use
    # ============================================================
    sw   x16, 0(x2)        # MEM[128] = 32
    lw   x17, 0(x2)        # x17 = 32
    add  x18, x17, x16     # x18 = 64

    # ============================================================
    # 5) Consecutive writes same rd, forwarding priority
    # ============================================================
    addi x19, x0, 1
    addi x19, x19, 1       # x19 = 2
    addi x19, x19, 1       # x19 = 3
    add  x22, x19, x0      # x22 = 3

    # ============================================================
    # 6) Branch not taken
    # ============================================================
    bne  x22, x19, BAD1    # not taken
    addi x23, x0, 23       # must execute

    # ============================================================
    # 7) Branch taken after load-use
    # ============================================================
    sw   x23, 8(x1)        # MEM[72] = 23
    lw   x24, 8(x1)        # x24 = 23
    beq  x24, x23, B2_OK   # taken, branch uses loaded value
    addi x20, x0, 123      # WRONG PATH
    addi x21, x0, 124      # WRONG PATH

B2_OK:
    # ============================================================
    # 8) JAL flush + PC+4
    # ============================================================
    jal  x25, JAL_OK       # x25 = PC+4
    addi x20, x0, 125      # WRONG PATH

JAL_OK:
    addi x26, x0, 26

    # ============================================================
    # 9) JALR with forwarding into branch/jump unit
    # ============================================================
    auipc x27, 0
    addi  x27, x27, 16     # target = JALR_OK
    jalr  x28, 0(x27)      # x28 = PC+4, jump to JALR_OK
    addi  x21, x0, 126     # WRONG PATH

JALR_OK:
    addi x29, x0, 29

    # ============================================================
    # 10) x0 write protection
    # ============================================================
    addi x0, x0, 99
    add  x30, x0, x29      # x30 = 29

    # ============================================================
    # 11) Shifts
    # ============================================================
    addi x31, x0, 1
    slli x31, x31, 5       # x31 = 32
    srli x30, x31, 2       # x30 = 8
    srai x29, x31, 1       # x29 = 16

    # ============================================================
    # 12) Signed / unsigned comparisons
    # ============================================================
    addi x7, x0, -1        # x7 = 0xFFFFFFFF
    slt  x8, x7, x0        # signed: -1 < 0 => x8 = 1
    sltu x9, x7, x0        # unsigned: FFFFFFFF < 0 => x9 = 0
    sltu x10, x0, x7       # unsigned: 0 < FFFFFFFF => x10 = 1

    # ============================================================
    # 13) Logic chain with forwarding
    # ============================================================
    addi x11, x0, 15       # 0xF
    addi x12, x0, 10       # 0xA
    and  x13, x11, x12     # x13 = 10
    or   x14, x13, x12     # x14 = 10
    xor  x15, x14, x11     # x15 = 5

    # ============================================================
    # 14) Store/load multiple addresses
    # ============================================================
    sw   x15, 0(x3)        # MEM[192] = 5
    lw   x16, 0(x3)        # x16 = 5
    addi x17, x16, 1       # x17 = 6

    sw   x17, 4(x3)        # MEM[196] = 6
    lw   x18, 4(x3)        # x18 = 6
    add  x19, x18, x17     # x19 = 12

    # ============================================================
    # 15) Branch taken with two wrong-path instructions
    # ============================================================
    beq  x19, x19, FINAL_OK
    addi x20, x0, 200      # WRONG PATH
    addi x21, x0, 201      # WRONG PATH

FINAL_OK:
    addi x22, x0, 77       # final marker

done:
    jal x0, done

BAD1:
    addi x20, x0, 250
    jal x0, done
