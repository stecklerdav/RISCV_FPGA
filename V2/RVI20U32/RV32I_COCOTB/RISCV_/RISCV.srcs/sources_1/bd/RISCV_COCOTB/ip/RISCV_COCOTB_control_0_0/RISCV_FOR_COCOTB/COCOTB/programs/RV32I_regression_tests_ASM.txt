
RV32I REGRESSION ASM TESTS
Mapa usado:
ROM  : 0x00000000
RAM  : 0x00002000
GPIO : 0x00003000
UART : 0x00003010
TIMER: 0x00003020

Convención:
- Si el test pasa, escribe 0xCAFE0000 + TC_ID en RAM[0x2000].
- Si falla, escribe 0xBAD00000 + TC_ID en RAM[0x2000].
- Todos los tests terminan en loop infinito.

====================================================================
TC-001_rv32i_alu_basic_signature.S
====================================================================
.option norvc
.section .text.start
.global _start

.equ RAM_BASE, 0x00002000

_start:
    li   x1, 10
    li   x2, 3

    add  x3, x1, x2        # 13
    li   x4, 13
    bne  x3, x4, fail

    sub  x3, x1, x2        # 7
    li   x4, 7
    bne  x3, x4, fail

    and  x3, x1, x2        # 10 & 3 = 2
    li   x4, 2
    bne  x3, x4, fail

    or   x3, x1, x2        # 10 | 3 = 11
    li   x4, 11
    bne  x3, x4, fail

    xor  x3, x1, x2        # 10 ^ 3 = 9
    li   x4, 9
    bne  x3, x4, fail

    sll  x3, x2, x2        # 3 << 3 = 24
    li   x4, 24
    bne  x3, x4, fail

    srl  x3, x1, x2        # 10 >> 3 = 1
    li   x4, 1
    bne  x3, x4, fail

    li   x5, -8
    srai x3, x5, 1         # -4
    li   x4, -4
    bne  x3, x4, fail

    slt  x3, x5, x1        # -8 < 10 = 1
    li   x4, 1
    bne  x3, x4, fail

    sltu x3, x5, x1        # unsigned -8 > 10 = 0
    li   x4, 0
    bne  x3, x4, fail

pass:
    li   x10, RAM_BASE
    li   x11, 0xCAFE0001
    sw   x11, 0(x10)
    j    done

fail:
    li   x10, RAM_BASE
    li   x11, 0xBAD00001
    sw   x11, 0(x10)

done:
    j    done
